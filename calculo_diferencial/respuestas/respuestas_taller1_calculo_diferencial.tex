\section{Límites con forma $0/0$}
\begin{enumerate}
    \item $\displaystyle \lim_{x\to 0} \frac{\sin(4x)}{x} = 4$.
    \item $\displaystyle \lim_{x\to 0} \frac{e^{2x}-1}{x} = 2$.
    \item $\displaystyle \lim_{x\to 0} \frac{1-\cos(2x)}{x^2} = 2$.
    \item $\displaystyle \lim_{x\to 0} \frac{\sqrt{1+x}-1}{x} = \frac{1}{2}$.
    \item $\displaystyle \lim_{x\to 2} \frac{x^2-4}{x-2} = 4$.
\end{enumerate}

\section{Límites con forma $\infty/\infty$}
\begin{enumerate}
    \item $\displaystyle \lim_{x\to \infty} \frac{\ln x}{x} = 0$.
    \item $\displaystyle \lim_{x\to \infty} \frac{x}{e^x} = 0$.
    \item $\displaystyle \lim_{x\to \infty} \frac{x^2}{e^x} = 0$.
    \item $\displaystyle \lim_{x\to \infty} \frac{5x-1}{2x+7} = \frac{5}{2}$.
\end{enumerate}

\section{Aplicaciones con varias derivadas}
\begin{enumerate}
    \item $\displaystyle \lim_{x\to 0} \frac{x-\sin x}{x^3} = \frac{1}{6}$.
    \item $\displaystyle \lim_{x\to 0} \frac{\tan x - x}{x^3} = \frac{1}{3}$.
    \item $\displaystyle \lim_{x\to 0} \frac{e^x - 1 - x}{x^2} = \frac{1}{2}$.
    \item $\displaystyle \lim_{x\to 0} \frac{\ln(1+x)}{x} = 1$.
    \item $\displaystyle \lim_{x\to \infty} \frac{\ln x}{\sqrt{x}} = 0$.
\end{enumerate}

\section{Ejercicios de aplicación}
\begin{enumerate}
    \item Si se interpreta la pendiente mediante el cociente incremental, se obtiene
    \[
    \lim_{x\to 0} \frac{\sin x / x - 1}{x} = 0,
    \]
    por lo que la pendiente buscada es $0$.

    \item Como $e^x$ crece más rápido que cualquier polinomio, se tiene
    \[
    \lim_{x\to\infty} \frac{e^x}{x^2} = \infty.
    \]

    \item Aplicando L'Hopital:
    \[
    \lim_{x\to 0} \frac{\ln(1+x)}{x}
    = \lim_{x\to 0} \frac{\frac{1}{1+x}}{1} = 1.
    \]
\end{enumerate}

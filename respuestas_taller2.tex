\section{Racionalización}

\begin{enumerate}
    \item $\frac{5}{\sqrt{3}} = \frac{5\sqrt{3}}{3}$
    
    \item $\frac{2}{\sqrt{5} + 1}$
    \begin{align*}
        &= \frac{2(\sqrt{5} - 1)}{(\sqrt{5} + 1)(\sqrt{5} - 1)}\\
        &= \frac{2(\sqrt{5} - 1)}{5 - 1}\\
        &= \frac{2(\sqrt{5} - 1)}{4}\\
        &= \frac{\sqrt{5} - 1}{2}
    \end{align*}
    
    \item $\frac{3}{\sqrt{7} - \sqrt{2}}$
    \begin{align*}
        &= \frac{3(\sqrt{7} + \sqrt{2})}{(\sqrt{7} - \sqrt{2})(\sqrt{7} + \sqrt{2})}\\
        &= \frac{3(\sqrt{7} + \sqrt{2})}{7 - 2}\\
        &= \frac{3(\sqrt{7} + \sqrt{2})}{5}
    \end{align*}
    
    \item $\frac{\sqrt{2} + 1}{\sqrt{2} - 1}$
    \begin{align*}
        &= \frac{(\sqrt{2} + 1)(\sqrt{2} + 1)}{(\sqrt{2} - 1)(\sqrt{2} + 1)}\\
        &= \frac{(\sqrt{2} + 1)^2}{2 - 1}\\
        &= \frac{2 + 2\sqrt{2} + 1}{1}\\
        &= 3 + 2\sqrt{2}
    \end{align*}
    
    \item $\frac{4}{\sqrt[3]{2}} = \frac{4\sqrt[3]{4}}{2} = 2\sqrt[3]{4}$
    
    \item $\frac{x + \sqrt{x}}{x - \sqrt{x}}$
    \begin{align*}
        &= \frac{(x + \sqrt{x})(x + \sqrt{x})}{(x - \sqrt{x})(x + \sqrt{x})}\\
        &= \frac{(x + \sqrt{x})^2}{x^2 - x}\\
        &= \frac{x^2 + 2x\sqrt{x} + x}{x(x - 1)}\\
        &= \frac{x^2 + x + 2x\sqrt{x}}{x(x - 1)}
    \end{align*}
    
    \item $\frac{6}{2\sqrt{3} + 3\sqrt{2}}$
    \begin{align*}
        &= \frac{6(2\sqrt{3} - 3\sqrt{2})}{(2\sqrt{3} + 3\sqrt{2})(2\sqrt{3} - 3\sqrt{2})}\\
        &= \frac{6(2\sqrt{3} - 3\sqrt{2})}{12 - 18}\\
        &= \frac{6(2\sqrt{3} - 3\sqrt{2})}{-6}\\
        &= -(2\sqrt{3} - 3\sqrt{2})\\
        &= 3\sqrt{2} - 2\sqrt{3}
    \end{align*}
\end{enumerate}

\section{Inecuaciones}

\begin{enumerate}
    \item $3x - 5 < 10$
    \begin{align*}
        3x &< 15\\
        x &< 5
    \end{align*}
    Solución: $(-\infty, 5)$
    
    \item $2(x - 3) \geq x + 4$
    \begin{align*}
        2x - 6 &\geq x + 4\\
        x &\geq 10
    \end{align*}
    Solución: $[10, \infty)$
    
    \item $-4 \leq 2x + 6 < 10$
    \begin{align*}
        -10 \leq 2x &< 4\\
        -5 \leq x &< 2
    \end{align*}
    Solución: $[-5, 2)$
    
    \item $x^2 - 5x + 6 > 0$
    \begin{align*}
        (x - 2)(x - 3) &> 0
    \end{align*}
    Solución: $(-\infty, 2) \cup (3, \infty)$
    
    \item $x^2 - 4x + 3 \leq 0$
    \begin{align*}
        (x - 1)(x - 3) &\leq 0
    \end{align*}
    Solución: $[1, 3]$
    
    \item $\frac{x + 2}{x - 3} > 0$\\
    Puntos críticos: $x = -2, x = 3$\\
    Solución: $(-\infty, -2) \cup (3, \infty)$
    
    \item $|2x - 1| < 5$
    \begin{align*}
        -5 < 2x - 1 &< 5\\
        -4 < 2x &< 6\\
        -2 < x &< 3
    \end{align*}
    Solución: $(-2, 3)$
    
    \item $\frac{2x - 1}{x + 3} \geq 1$
    \begin{align*}
        \frac{2x - 1}{x + 3} - 1 &\geq 0\\
        \frac{2x - 1 - x - 3}{x + 3} &\geq 0\\
        \frac{x - 4}{x + 3} &\geq 0
    \end{align*}
    Solución: $(-\infty, -3) \cup [4, \infty)$
\end{enumerate}

\section{Operaciones con Conjuntos - Demostración}

\begin{enumerate}
    \item \textbf{Ley distributiva:} $A \cup (B \cap C) = (A \cup B) \cap (A \cup C)$
    
    \textbf{Demostración:} Sea $x \in A \cup (B \cap C)$. Entonces:
    \begin{itemize}
        \item Si $x \in A$, entonces $x \in A \cup B$ y $x \in A \cup C$, por lo tanto $x \in (A \cup B) \cap (A \cup C)$.
        \item Si $x \in B \cap C$, entonces $x \in B$ y $x \in C$, luego $x \in A \cup B$ y $x \in A \cup C$, así $x \in (A \cup B) \cap (A \cup C)$.
    \end{itemize}
    La inclusión contraria se demuestra de manera similar.
    
    \item \textbf{Ley de De Morgan:} $(A \cup B)^c = A^c \cap B^c$
    
    \textbf{Demostración:} $x \in (A \cup B)^c \Leftrightarrow x \notin A \cup B \Leftrightarrow x \notin A \text{ y } x \notin B \Leftrightarrow x \in A^c \text{ y } x \in B^c \Leftrightarrow x \in A^c \cap B^c$
    
    \item \textbf{Diferencia:} $A - B = A \cap B^c$
    
    \textbf{Demostración:} $x \in A - B \Leftrightarrow x \in A \text{ y } x \notin B \Leftrightarrow x \in A \text{ y } x \in B^c \Leftrightarrow x \in A \cap B^c$
\end{enumerate}

\section{Problemas de Aplicación con Conjuntos}

\begin{enumerate}
    \item Sean: $F$ = fútbol, $B$ = baloncesto, $N$ = natación
    
    Datos: $|F| = 45$, $|B| = 35$, $|N| = 30$, $|F \cap B| = 15$, $|F \cap N| = 12$, $|B \cap N| = 10$, $|F \cap B \cap N| = 5$
    
    \begin{enumerate}
        \item Solo fútbol: $|F| - |F \cap B| - |F \cap N| + |F \cap B \cap N| = 45 - 15 - 12 + 5 = 23$
        
        \item Exactamente dos deportes:
        \begin{align*}
            &(|F \cap B| - |F \cap B \cap N|) + (|F \cap N| - |F \cap B \cap N|) + (|B \cap N| - |F \cap B \cap N|)\\
            &= (15 - 5) + (12 - 5) + (10 - 5) = 10 + 7 + 5 = 22
        \end{align*}
        
        \item Ningún deporte:
        \begin{align*}
            &100 - [|F| + |B| + |N| - |F \cap B| - |F \cap N| - |B \cap N| + |F \cap B \cap N|]\\
            &= 100 - [45 + 35 + 30 - 15 - 12 - 10 + 5]\\
            &= 100 - 78 = 22
        \end{align*}
    \end{enumerate}
    
    \item Sean: $I$ = inglés, $F$ = francés, $A$ = alemán
    
    Datos: $|I| = 32$, $|F| = 28$, $|A| = 25$, $|I \cap F| = 18$, $|I \cap A| = 15$, $|F \cap A| = 14$, $|I \cap F \cap A| = 10$
    
    \begin{align*}
        &\text{Hablan al menos uno} = |I| + |F| + |A| - |I \cap F| - |I \cap A| - |F \cap A| + |I \cap F \cap A|\\
        &= 32 + 28 + 25 - 18 - 15 - 14 + 10 = 48\\
        &\text{Ningún idioma} = 50 - 48 = 2
    \end{align*}
\end{enumerate}

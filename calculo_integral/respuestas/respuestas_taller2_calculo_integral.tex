\section{Integrales Definidas y Propiedades}
\begin{enumerate}
    \item $\int_{-2}^{2} (x^5+3x^2)\,dx = 16$.
    \item $\int_0^3 |x-1|\,dx = \frac{5}{2}$.
    \item $\int_1^4 (2x-3)\,dx = 6$.
    \item $\int_0^{\pi} \cos x\,dx = 0$.
    \item $\int_0^6 f(x)\,dx = 5 + (-3) = 2$.
\end{enumerate}

\section{Teorema Fundamental del Calculo}
\begin{enumerate}
    \item Si $F(x)=\int_1^x (t^2+2t)\,dt$, entonces
    $$F'(x)=x^2+2x, \qquad F(3)=\int_1^3 (t^2+2t)\,dt=\frac{50}{3}.$$

    \item Si $G(x)=\int_x^{2x} \ln(1+t^2)\,dt$, entonces
    $$G'(x)=2\ln(1+4x^2)-\ln(1+x^2).$$

    \item
    $$\frac{d}{dx}\left(\int_0^{x^3} \sqrt{1+t^4}\,dt\right)=\sqrt{1+x^{12}}\cdot 3x^2.$$

    \item Para $H(x)=\int_{\sin x}^{\cos x} (1+t^3)\,dt$,
    $$H'(x)=-(1+\cos^3 x)\sin x-(1+\sin^3 x)\cos x.$$
\end{enumerate}

\section{Area Entre Curvas}
\begin{enumerate}
    \item Entre $y=x$ y $y=x^2$:
    $$A=\int_0^1 (x-x^2)\,dx=\frac{1}{6}.$$

    \item Entre $y=4-x^2$ y eje $x$:
    $$A=\int_{-2}^{2}(4-x^2)\,dx=\frac{32}{3}.$$

    \item Entre $y=2x+3$ y $y=x^2$ (intersecciones en $x=-1,3$):
    $$A=\int_{-1}^{3}(2x+3-x^2)\,dx=\frac{32}{3}.$$

    \item Entre $y=\sin x$ y $y=\cos x$ en $[0,\pi/2]$:
    $$A=\int_0^{\pi/4}(\cos x-\sin x)\,dx+\int_{\pi/4}^{\pi/2}(\sin x-\cos x)\,dx=2(\sqrt{2}-1).$$
\end{enumerate}

\section{Volumenes por Discos y Arandelas}
\begin{enumerate}
    \item Giro de $y=x^2$, $0\le x\le 2$, alrededor de eje $x$:
    $$V=\pi\int_0^2 (x^2)^2\,dx=\frac{32\pi}{5}.$$

    \item Giro de $y=\sqrt{x}$, $0\le x\le 4$, alrededor de eje $x$:
    $$V=\pi\int_0^4 (\sqrt{x})^2\,dx=\pi\int_0^4 x\,dx=8\pi.$$

    \item Giro de la region bajo $y=4-x$ en torno al eje $y$:
    $$V=\pi\int_0^4 (4-y)^2\,dy=\frac{64\pi}{3}.$$
\end{enumerate}

\section{Longitud de Arco y Promedio}
\begin{enumerate}
    \item Para $y=\frac{x^2}{2}$ en $[0,1]$, $y'=x$:
    $$L=\int_0^1\sqrt{1+x^2}\,dx=\frac{1}{2}\left(\sqrt{2}+\ln(1+\sqrt{2})\right).$$

    \item Valor promedio de $f(x)=x^2+1$ en $[0,3]$:
    $$f_{\text{prom}}=\frac{1}{3}\int_0^3(x^2+1)\,dx=4.$$

    \item Valor promedio de $f(x)=\sin x$ en $[0,\pi]$:
    $$f_{\text{prom}}=\frac{1}{\pi}\int_0^{\pi}\sin x\,dx=\frac{2}{\pi}.$$
\end{enumerate}

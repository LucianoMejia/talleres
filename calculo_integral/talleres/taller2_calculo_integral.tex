\section*{Instrucciones}
Resuelva cada ejercicio justificando cada paso. Si usa propiedades de simetria o teoremas, indiquelo explicitamente.

\section{Integrales Definidas y Propiedades}

\begin{enumerate}
    \item $\int_{-2}^{2} (x^5+3x^2)\,dx$
    \item $\int_0^3 |x-1|\,dx$
    \item $\int_1^4 (2x-3)\,dx$
    \item $\int_0^{\pi} \cos x\,dx$
    \item Si $\int_0^2 f(x)\,dx=5$ y $\int_2^6 f(x)\,dx=-3$, halle $\int_0^6 f(x)\,dx$.
\end{enumerate}

\section{Teorema Fundamental del Calculo}

\begin{enumerate}
    \item Si $F(x)=\int_1^x (t^2+2t)\,dt$, encuentre $F'(x)$ y $F(3)$.
    \item Si $G(x)=\int_{x}^{2x} \ln(1+t^2)\,dt$, calcule $G'(x)$.
    \item Determine $\frac{d}{dx}\left(\int_0^{x^3} \sqrt{1+t^4}\,dt\right)$.
    \item Sea $H(x)=\int_{\sin x}^{\cos x} (1+t^3)\,dt$. Halle $H'(x)$.
\end{enumerate}

\section{Area Entre Curvas}

\begin{enumerate}
    \item Calcule el area encerrada entre $y=x$ y $y=x^2$.
    \item Halle el area entre $y=4-x^2$ y el eje $x$.
    \item Determine el area limitada por $y=2x+3$ y $y=x^2$.
    \item Encuentre el area entre $y=\sin x$ y $y=\cos x$ en $[0,\pi/2]$.
\end{enumerate}

\section{Volumenes por Discos y Arandelas}

\begin{enumerate}
    \item Genere el volumen al girar la region bajo $y=x^2$, $0\le x\le 2$, alrededor del eje $x$.
    \item Genere el volumen al girar la region entre $y=\sqrt{x}$ y $y=0$, $0\le x\le 4$, alrededor del eje $x$.
    \item Calcule el volumen del solido al girar la region entre $y=4-x$ y $y=0$, $0\le x\le 4$, alrededor del eje $y$ (metodo de arandelas en funcion de $y$).
\end{enumerate}

\section{Longitud de Arco y Promedio}

\begin{enumerate}
    \item Encuentre la longitud de arco de $y=\frac{x^2}{2}$ en $[0,1]$.
    \item Calcule el valor promedio de $f(x)=x^2+1$ en $[0,3]$.
    \item Halle el valor promedio de $f(x)=\sin x$ en $[0,\pi]$.
\end{enumerate}

\section*{Respuestas}

\subsection*{Sistemas de ecuaciones y forma matricial}
\begin{enumerate}
    \item
    \[
    A = \begin{pmatrix}
    2 & 3 \\
    1 & -1
    \end{pmatrix},
    \quad
    X = \begin{pmatrix} x \\ y \end{pmatrix},
    \quad
    b = \begin{pmatrix} 7 \\ 1 \end{pmatrix}
    \]
    \[
    [A|b] = \left[\begin{array}{cc|c}
    2 & 3 & 7 \\
    1 & -1 & 1
    \end{array}\right]
    \]

    \item
    \[
    A = \begin{pmatrix}
    1 & 1 & 1 \\
    2 & -1 & 1 \\
    1 & 2 & -1
    \end{pmatrix},
    \quad
    X = \begin{pmatrix} x \\ y \\ z \end{pmatrix},
    \quad
    b = \begin{pmatrix} 6 \\ 3 \\ 4 \end{pmatrix}
    \]
    \[
    [A|b] = \left[\begin{array}{ccc|c}
    1 & 1 & 1 & 6 \\
    2 & -1 & 1 & 3 \\
    1 & 2 & -1 & 4
    \end{array}\right]
    \]
\end{enumerate}

\subsection*{Eliminación de Gauss}
\begin{enumerate}
    \item
    \[
    \begin{array}{cc|c}
    1 & 2 & 5 \\
    3 & -1 & 1
    \end{array}
    \xrightarrow{F_2-3F_1}
    \begin{array}{cc|c}
    1 & 2 & 5 \\
    0 & -7 & -14
    \end{array}
    \]
    \[
    -7y = -14 \Rightarrow y = 2
    \]
    \[
    x + 2(2) = 5 \Rightarrow x = 1
    \]
    \[
    \boxed{(x,y)=(1,2)}
    \]

    \item
    \[
    \begin{array}{ccc|c}
    1 & 1 & 1 & 6 \\
    2 & -1 & 3 & 14 \\
    3 & 2 & -1 & 2
    \end{array}
    \]
    Realizando operaciones elementales:
    \[
    \begin{array}{ccc|c}
    1 & 1 & 1 & 6 \\
    0 & -3 & 1 & 2 \\
    0 & -1 & -4 & -16
    \end{array}
    \]
    \[
    \begin{array}{ccc|c}
    1 & 1 & 1 & 6 \\
    0 & -3 & 1 & 2 \\
    0 & 0 & -\frac{11}{3} & -\frac{46}{3}
    \end{array}
    \]
    \[
    z = 4, \quad y = 2, \quad x = 0
    \]
    \[
    \boxed{(x,y,z)=(0,2,4)}
    \]
\end{enumerate}

\subsection*{Determinantes}
\begin{enumerate}
    \item
    \[
    \det(A) = 2(7)-3(5)=14-15=-1
    \]
    \[
    \boxed{\det(A)=-1}
    \]

    \item
    \[
    \det(B) = 1(4)-2(3)=4-6=-2
    \]
    \[
    \boxed{\det(B)=-2}
    \]

    \item
    \[
    \det(C)=3\begin{vmatrix}0&1\\4&3\end{vmatrix}-1\begin{vmatrix}2&1\\1&3\end{vmatrix}+2\begin{vmatrix}2&0\\1&4\end{vmatrix}
    \]
    \[
    =3(0\cdot3-1\cdot4)-1(2\cdot3-1\cdot1)+2(2\cdot4-0\cdot1)
    \]
    \[
    =3(-4)-1(5)+2(8)=-12-5+16=-1
    \]
    \[
    \boxed{\det(C)=-1}
    \]
\end{enumerate}

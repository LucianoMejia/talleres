\section{Jerarquía de Operaciones}

\begin{enumerate}
    \item $5 + 3 \times 2^2 - 8 \div 4$
    \begin{align*}
        &= 5 + 3 \times 4 - 8 \div 4\\
        &= 5 + 12 - 2\\
        &= 15
    \end{align*}
    
    \item $\left(12 - 3\right) \times 2 + 5^2 \div 5$
    \begin{align*}
        &= 9 \times 2 + 25 \div 5\\
        &= 18 + 5\\
        &= 23
    \end{align*}
    
    \item $\frac{15 + 3 \times 4}{9 - 2^2}$
    \begin{align*}
        &= \frac{15 + 12}{9 - 4}\\
        &= \frac{27}{5}
    \end{align*}
    
    \item $2 \times \left[8 - 3 \times (4 - 1)\right] + 5$
    \begin{align*}
        &= 2 \times [8 - 3 \times 3] + 5\\
        &= 2 \times [8 - 9] + 5\\
        &= 2 \times (-1) + 5\\
        &= -2 + 5\\
        &= 3
    \end{align*}
    
    \item $\frac{6 \times 4 - 2^3}{3 + 1} + 7 \times 2$
    \begin{align*}
        &= \frac{24 - 8}{4} + 14\\
        &= \frac{16}{4} + 14\\
        &= 4 + 14\\
        &= 18
    \end{align*}
\end{enumerate}

\section{Homogenización}

\begin{enumerate}
    \item $5x + 3y - 2x + 7y - x$
    \begin{align*}
        &= (5x - 2x - x) + (3y + 7y)\\
        &= 2x + 10y
    \end{align*}
    
    \item $4a^2 - 3ab + 2b^2 + 5ab - a^2 + 3b^2$
    \begin{align*}
        &= (4a^2 - a^2) + (-3ab + 5ab) + (2b^2 + 3b^2)\\
        &= 3a^2 + 2ab + 5b^2
    \end{align*}
    
    \item $\frac{2x}{3} + \frac{x}{4} - \frac{5x}{6}$
    \begin{align*}
        &= \frac{8x + 3x - 10x}{12}\\
        &= \frac{x}{12}
    \end{align*}
    
    \item $3(x + 2) - 2(x - 1) + 4(2x + 3)$
    \begin{align*}
        &= 3x + 6 - 2x + 2 + 8x + 12\\
        &= (3x - 2x + 8x) + (6 + 2 + 12)\\
        &= 9x + 20
    \end{align*}
    
    \item $\frac{x + 1}{2} + \frac{2x - 3}{4} - \frac{x - 2}{3}$
    \begin{align*}
        &= \frac{6(x + 1) + 3(2x - 3) - 4(x - 2)}{12}\\
        &= \frac{6x + 6 + 6x - 9 - 4x + 8}{12}\\
        &= \frac{8x + 5}{12}
    \end{align*}
\end{enumerate}

\section{Factorización}

\begin{enumerate}
    \item $x^2 + 5x + 6 = (x + 2)(x + 3)$
    
    \item $4x^2 - 9 = (2x)^2 - 3^2 = (2x - 3)(2x + 3)$
    
    \item $x^2 - 6x + 9 = (x - 3)^2$
    
    \item $2x^2 + 7x + 3 = (2x + 1)(x + 3)$
    
    \item $x^3 - 8 = x^3 - 2^3 = (x - 2)(x^2 + 2x + 4)$
    
    \item $6x^2 - 13x - 5 = (3x + 1)(2x - 5)$
    
    \item $x^2 + 6x + 9 - y^2 = (x + 3)^2 - y^2 = (x + 3 - y)(x + 3 + y)$
    
    \item $ax + ay + bx + by = a(x + y) + b(x + y) = (a + b)(x + y)$
\end{enumerate}

\section{Ecuaciones}

\begin{enumerate}
    \item $3x + 7 = 22$
    \begin{align*}
        3x &= 15\\
        x &= 5
    \end{align*}
    
    \item $\frac{2x - 3}{4} = \frac{x + 1}{2}$
    \begin{align*}
        2(2x - 3) &= 4(x + 1)\\
        4x - 6 &= 4x + 4\\
        -6 &= 4
    \end{align*}
    No tiene solución (ecuación inconsistente)
    
    \item $x^2 - 5x + 6 = 0$
    \begin{align*}
        (x - 2)(x - 3) &= 0\\
        x = 2 \quad &\text{o} \quad x = 3
    \end{align*}
    
    \item $2x^2 + 3x - 2 = 0$
    \begin{align*}
        (2x - 1)(x + 2) &= 0\\
        x = \frac{1}{2} \quad &\text{o} \quad x = -2
    \end{align*}
    
    \item $\frac{1}{x - 1} + \frac{2}{x + 1} = 3$
    \begin{align*}
        (x + 1) + 2(x - 1) &= 3(x - 1)(x + 1)\\
        x + 1 + 2x - 2 &= 3(x^2 - 1)\\
        3x - 1 &= 3x^2 - 3\\
        3x^2 - 3x - 2 &= 0\\
        x &= \frac{3 \pm \sqrt{9 + 24}}{6} = \frac{3 \pm \sqrt{33}}{6}
    \end{align*}
    
    \item $\sqrt{2x + 5} = 3$
    \begin{align*}
        2x + 5 &= 9\\
        2x &= 4\\
        x &= 2
    \end{align*}
    
    \item $|3x - 4| = 8$
    \begin{align*}
        3x - 4 = 8 \quad &\text{o} \quad 3x - 4 = -8\\
        x = 4 \quad &\text{o} \quad x = -\frac{4}{3}
    \end{align*}
    
    \item $(x - 2)^2 = 16$
    \begin{align*}
        x - 2 = \pm 4\\
        x = 6 \quad &\text{o} \quad x = -2
    \end{align*}
\end{enumerate}

\section{Problemas de Aplicación}

\begin{enumerate}
    \item \textbf{Solución:} Sean $n$ y $n+1$ los números consecutivos.
    \begin{align*}
        n + (n + 1) &= 47\\
        2n + 1 &= 47\\
        2n &= 46\\
        n &= 23
    \end{align*}
    Los números son 23 y 24.
    
    \item \textbf{Solución:} Sea $x$ el ancho, entonces el largo es $2x$.
    \begin{align*}
        2(x + 2x) &= 36\\
        6x &= 36\\
        x &= 6
    \end{align*}
    Dimensiones: ancho = 6 m, largo = 12 m.
    
    \item \textbf{Solución:} Sea $x$ la edad de Juan, entonces Pedro tiene $3x$ años.
    \begin{align*}
        (x + 8) + (3x + 8) &= 56\\
        4x + 16 &= 56\\
        4x &= 40\\
        x &= 10
    \end{align*}
    Juan tiene 10 años y Pedro tiene 30 años.
    
    \item \textbf{Solución:} La primera llave llena $\frac{1}{6}$ del tanque por hora, la segunda $\frac{1}{4}$.
    \begin{align*}
        \frac{1}{6} + \frac{1}{4} &= \frac{2 + 3}{12} = \frac{5}{12}\\
        t &= \frac{12}{5} = 2.4 \text{ horas}
    \end{align*}
    El tanque se llenará en 2 horas y 24 minutos.
    
    \item \textbf{Solución:} Sea $n$ el número de artículos y $p$ el precio por artículo.
    \begin{align*}
        np &= 240\\
        (n + 4)(p - 2) &= 240\\
        np - 2n + 4p - 8 &= 240\\
        240 - 2n + 4 \cdot \frac{240}{n} - 8 &= 240\\
        -2n + \frac{960}{n} - 8 &= 0\\
        -2n^2 - 8n + 960 &= 0\\
        n^2 + 4n - 480 &= 0\\
        (n + 24)(n - 20) &= 0
    \end{align*}
    Como $n > 0$, entonces $n = 20$. Compró 20 artículos.
\end{enumerate}
